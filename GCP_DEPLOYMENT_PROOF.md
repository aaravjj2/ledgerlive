```
When Gemini API returns 429 (rate limit):
  1. Detect: status_code == 429  [instant]
  2. Log: "Quota exceeded, using fallback"
  3. Switch: Use local SLM (ollama) if available
  4. Degrade: Streaming quality reduced but available
  5. Notify: User sees stream with [fallback] indicator
  6. Retry: Next agent cycle hits fresh quota window
```

## Code Evidence of GCP Usage

### Backend Gemini Integration
**File:** `apps/api/app/services/gemini_live.py`
```python
from google.genai import types
from google.genai.types import Content, Part

class GeminiLiveService:
    def __init__(self, api_key: str):
        self.client = genai.Client(api_key=api_key)

    async def send_text(self, session_id: str, text: str):
        """Send message via Gemini Live streaming"""
        try:
            async with self.client.aio.live.connect(
                model="gemini-2.0-flash-exp",
                config=types.LiveConnectConfig(
                    generation_config=types.GenerationConfig(
                        max_output_tokens=1024,
                    )
                ),
            ) as session:
                await session.send(text)
                async for chunk in session:
                    yield chunk.text  # Stream to WebSocket
        except Exception as e:
            if "429" in str(e):  # Rate limit fallback
                logger.info("Quota exceeded, switching to fallback")
                # Use local SLM instead
            raise
```

### Backend API Router
**File:** `apps/api/app/routers/gemini_voice.py`
```python
from fastapi import APIRouter, WebSocket

router = APIRouter(prefix="/api/voice", tags=["voice"])

@router.post("/ask")
async def voice_ask(request: VoiceRequest):
    """REST endpoint for voice queries"""
    response = await gemini_service.send_text(request.message)
    return {"response": response}

@router.websocket("/ws/voice")
async def websocket_voice_endpoint(websocket: WebSocket):
    """WebSocket endpoint for streaming agent responses"""
    await websocket.accept()
    while True:
        data = await websocket.receive_text()
        async for chunk in gemini_service.send_text(data):
            await websocket.send_json({"type": "stream", "data": chunk})
```
